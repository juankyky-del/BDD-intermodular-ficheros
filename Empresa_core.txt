DROP DATABASE IF EXISTS empresa_core;
CREATE DATABASE empresa_core CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;
USE empresa_core;

-- =====================================================================================
-- 1. TABLAS BASE (Entidades principales)
-- =====================================================================================

-- 1. DEPARTAMENTOS
CREATE TABLE departamentos (
    id_departamento INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(100) NOT NULL UNIQUE,
    ubicacion VARCHAR(100) NOT NULL,
    presupuesto_anual DECIMAL(12,2) DEFAULT 0.00
);

-- 2. PUESTOS (Catálogo salarial y roles organizativos)
CREATE TABLE puestos (
    id_puesto INT AUTO_INCREMENT PRIMARY KEY,
    titulo VARCHAR(100) NOT NULL UNIQUE,
    salario_minimo DECIMAL(10,2) NOT NULL,
    salario_maximo DECIMAL(10,2) NOT NULL,
    CONSTRAINT chk_salario CHECK (salario_maximo >= salario_minimo)
);

-- 3. EMPLEADOS
CREATE TABLE empleados (
    id_empleado INT AUTO_INCREMENT PRIMARY KEY,
    dni VARCHAR(15) NOT NULL UNIQUE,
    nombre VARCHAR(50) NOT NULL,
    apellidos VARCHAR(100) NOT NULL,
    email_corporativo VARCHAR(100) NOT NULL UNIQUE,
    id_departamento INT,
    id_puesto INT,
    fecha_contratacion DATE NOT NULL,
    estado ENUM('ACTIVO', 'BAJA_TEMPORAL', 'DESPEDIDO', 'EXCEDENCIA') DEFAULT 'ACTIVO',
    FOREIGN KEY (id_departamento) REFERENCES departamentos(id_departamento) ON DELETE RESTRICT,
    FOREIGN KEY (id_puesto) REFERENCES puestos(id_puesto) ON DELETE RESTRICT
);

-- 4. CREDENCIALES DE ACCESO (Sistema de Intranet)
CREATE TABLE credenciales_acceso (
    id_credencial INT AUTO_INCREMENT PRIMARY KEY,
    id_empleado INT UNIQUE NOT NULL,
    username VARCHAR(50) NOT NULL UNIQUE,
    password_hash VARCHAR(255) NOT NULL,
    rol_sistema ENUM('ADMIN', 'RRHH', 'SUPERVISOR', 'EMPLEADO_BASE') DEFAULT 'EMPLEADO_BASE',
    fecha_ultimo_acceso DATETIME,
    cuenta_bloqueada BOOLEAN DEFAULT FALSE,
    FOREIGN KEY (id_empleado) REFERENCES empleados(id_empleado) ON DELETE CASCADE
);

-- 5. HISTORIAL SALARIAL (Trazabilidad para transacciones)
CREATE TABLE historial_salarial (
    id_historial INT AUTO_INCREMENT PRIMARY KEY,
    id_empleado INT NOT NULL,
    salario_anual DECIMAL(10,2) NOT NULL,
    fecha_inicio DATE NOT NULL,
    fecha_fin DATE NULL, -- NULL significa que es el salario actual vigente
    motivo_cambio VARCHAR(255) NOT NULL,
    FOREIGN KEY (id_empleado) REFERENCES empleados(id_empleado) ON DELETE CASCADE
);

-- 6. PROYECTOS (Iniciativas de la empresa)
CREATE TABLE proyectos (
    id_proyecto INT AUTO_INCREMENT PRIMARY KEY,
    nombre_proyecto VARCHAR(150) NOT NULL UNIQUE,
    id_departamento_responsable INT NOT NULL,
    presupuesto_asignado DECIMAL(12,2) NOT NULL,
    fecha_inicio DATE NOT NULL,
    fecha_fin_estimada DATE NOT NULL,
    fecha_cierre_real DATE NULL,
    estado ENUM('PLANIFICACION', 'EN_CURSO', 'PAUSADO', 'FINALIZADO', 'CANCELADO') DEFAULT 'PLANIFICACION',
    FOREIGN KEY (id_departamento_responsable) REFERENCES departamentos(id_departamento) ON DELETE RESTRICT
);

-- 7. EQUIPOS INFORMATICOS (Inventario de hardware de la empresa)
CREATE TABLE equipos_informaticos (
    id_equipo INT AUTO_INCREMENT PRIMARY KEY,
    numero_serie VARCHAR(100) NOT NULL UNIQUE,
    tipo_dispositivo ENUM('PORTATIL', 'SOBREMESA', 'MOVIL', 'MONITOR', 'PERIFERICO') NOT NULL,
    marca_modelo VARCHAR(150) NOT NULL,
    fecha_adquisicion DATE NOT NULL,
    id_empleado_asignado INT NULL, -- NULL indica que está en stock/almacén
    estado ENUM('OPERATIVO', 'EN_REPARACION', 'OBSOLETO', 'PERDIDO') DEFAULT 'OPERATIVO',
    FOREIGN KEY (id_empleado_asignado) REFERENCES empleados(id_empleado) ON DELETE SET NULL
);

-- 8. INCIDENCIAS SOPORTE (Ticketing / HelpDesk)
CREATE TABLE incidencias_soporte (
    id_incidencia INT AUTO_INCREMENT PRIMARY KEY,
    id_empleado_reporta INT NOT NULL,
    id_equipo_afectado INT NULL,
    descripcion_problema TEXT NOT NULL,
    fecha_reporte DATETIME DEFAULT CURRENT_TIMESTAMP,
    fecha_resolucion DATETIME NULL,
    estado ENUM('ABIERTA', 'EN_PROGRESO', 'ESPERANDO_PIEZAS', 'CERRADA') DEFAULT 'ABIERTA',
    resolucion TEXT NULL,
    FOREIGN KEY (id_empleado_reporta) REFERENCES empleados(id_empleado) ON DELETE CASCADE,
    FOREIGN KEY (id_equipo_afectado) REFERENCES equipos_informaticos(id_equipo) ON DELETE SET NULL
);

-- 9. FORMACIONES Y CURSOS (Catálogo de capacitación)
CREATE TABLE formaciones_cursos (
    id_curso INT AUTO_INCREMENT PRIMARY KEY,
    nombre_curso VARCHAR(200) NOT NULL,
    proveedor VARCHAR(100) NOT NULL,
    horas_duracion INT NOT NULL,
    coste_por_empleado DECIMAL(8,2) NOT NULL DEFAULT 0.00
);

-- 10. REGISTRO DE AUDITORIA (Logs de seguridad para ASIR)
CREATE TABLE registro_auditoria (
    id_log INT AUTO_INCREMENT PRIMARY KEY,
    id_credencial_autor INT NULL,
    tabla_afectada VARCHAR(50) NOT NULL,
    accion ENUM('INSERT', 'UPDATE', 'DELETE', 'LOGIN_FALLIDO', 'ALERTA_SEGURIDAD') NOT NULL,
    fecha_hora DATETIME DEFAULT CURRENT_TIMESTAMP,
    detalles TEXT NOT NULL,
    FOREIGN KEY (id_credencial_autor) REFERENCES credenciales_acceso(id_credencial) ON DELETE SET NULL
);

-- =====================================================================================
-- 2. TABLAS DE RELACIÓN (N:M)
-- =====================================================================================

-- 11. ASIGNACIONES DE PROYECTOS (Relación Empleados <-> Proyectos)
CREATE TABLE asignaciones_proyectos (
    id_proyecto INT NOT NULL,
    id_empleado INT NOT NULL,
    rol_en_proyecto VARCHAR(100) NOT NULL,
    horas_semanales INT NOT NULL DEFAULT 40,
    fecha_asignacion DATE NOT NULL,
    fecha_desasignacion DATE NULL,
    PRIMARY KEY (id_proyecto, id_empleado, fecha_asignacion),
    FOREIGN KEY (id_proyecto) REFERENCES proyectos(id_proyecto) ON DELETE CASCADE,
    FOREIGN KEY (id_empleado) REFERENCES empleados(id_empleado) ON DELETE CASCADE
);

-- 12. ASISTENCIAS A FORMACION (Relación Empleados <-> Cursos)
CREATE TABLE asistencias_formacion (
    id_curso INT NOT NULL,
    id_empleado INT NOT NULL,
    fecha_inscripcion DATE NOT NULL,
    estado ENUM('INSCRITO', 'EN_CURSO', 'COMPLETADO', 'ABANDONADO') DEFAULT 'INSCRITO',
    calificacion_final DECIMAL(4,2) NULL,
    PRIMARY KEY (id_curso, id_empleado),
    FOREIGN KEY (id_curso) REFERENCES formaciones_cursos(id_curso) ON DELETE CASCADE,
    FOREIGN KEY (id_empleado) REFERENCES empleados(id_empleado) ON DELETE CASCADE
);
